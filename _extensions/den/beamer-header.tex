% === Title Page ===
% Emblem and institution name are part of title.jpg. Title, rule and date
% are placed below them: the title grows upwards from the rule, so one to
% three lines never reach the emblem; subtitle and author sit under the rule.
\newlength{\dentitlewidth}
\setbeamertemplate{title page}{%
  \settowidth{\dentitlewidth}{\denheadfont\bfseries\densize{24}\inserttitle}%
  \begin{tikzpicture}[remember picture, overlay]
    \node[anchor=north west, inner sep=0pt] at (current page.north west)
      {\includegraphics[width=\paperwidth, height=\paperheight, keepaspectratio=false]{title.jpg}};
    \node[anchor=south, text width=7.7in, align=center, inner sep=0pt, text=white]
      at ([yshift=-4.0in]current page.north)
      {\denheadfont\bfseries
       \ifdim\dentitlewidth>23.1in\densize{20}\else\densize{24}\fi
       \dentitlesetup\baselineskip=1.08\baselineskip\inserttitle\par};
    \draw[white, line width=1.5pt] ([xshift=-3.97in, yshift=-4.09in]current page.north) -- ++(7.94in, 0);
    \node[anchor=north, text width=9in, align=center, inner sep=0pt, text=white]
      at ([yshift=-4.3in]current page.north)
      {{\denheadfont\bfseries\densize{16}\insertsubtitle\par}\vspace{4pt}%
       {\denheadfont\densize{14}\insertauthor\par}};
    \node[anchor=center, inner sep=0pt, text=white] at ([yshift=-5.33in]current page.north)
      {\denheadfont\bfseries\densize{16}\insertdate};
  \end{tikzpicture}%
}

% === Footer: Page Number ===
% 1 cm brown square in the bottom-right corner, numbered from the first
% slide after the title. Plain frames (title, dividers, ending) show none.

  \ifnum\insertframenumber>1
    \begin{tikzpicture}[remember picture, overlay]
      \node[anchor=south east, fill=denpage, text=white, inner sep=0pt,
            minimum width=1cm, minimum height=1cm, font=\bfseries\densize{12}]
        at (current page.south east) {\the\numexpr\insertframenumber-1\relax};
    \end{tikzpicture}%
  \fi
}

% === Source and Note ===
% Used by source-note.lua for `::: {.source}` and `::: {.note}`. The source
% sits on the bottom line, right-aligned against the page number; the note
% sits on the line above it, at the left text margin.
\newcommand{\densource}[2]{%
  \begin{tikzpicture}[remember picture, overlay]
    \node[anchor=base east, inner sep=0pt, align=right, font=\densize{12}]
      at ([xshift=-0.5in, yshift=0.13in]current page.south east) {\textbf{#1:} #2};
  \end{tikzpicture}%
}
\newcommand{\dennote}[2]{%
  \begin{tikzpicture}[remember picture, overlay]
    \node[anchor=south west, inner sep=0pt, text width=12.1in, align=left, font=\densize{12}]
      at ([xshift=0.5in, yshift=0.33in]current page.south west) {\textbf{#1:} #2};
  \end{tikzpicture}%
}

% === Slot Grid and Cards ===
% Used by cols.lua for `:::: {.cols n=3}`. Each slot is typeset into one of
% the \denslotA..L boxes at a fixed width and height, then the boxes are set
% out row by row. cols.lua chooses one style per slot:
%   dencard        cream box, orange border (an untitled .card)
%   denheadcard    coloured header bar over a white body (.card with title)
%   densoftcard    cream box with a brown heading inside (.card .soft)
%   denfillcard    whole card in one colour, white text (.card .fill)
%   denbanner      a bar of text (.banner)
%   denplain       invisible box for text or a figure (.plain)
\tcbset{
  denslot/.style={
    nobeforeafter, valign=top, halign=flush left,
    before upper={\setlength{\parskip}{0.3em}\setlength{\leftmargini}{1.1em}},
  },
  dencard/.style={
    denslot, enhanced, colback=dencream, colframe=denorange, coltext=dendark,
    boxrule=0.75pt, arc=0.12in, boxsep=0pt,
    left=0.15in, right=0.15in, top=0.1in, bottom=0.1in,
  },
  denheadcard/.style={
    denslot, enhanced, colback=white, colframe=#1, colbacktitle=#1, coltitle=white,
    boxrule=0.75pt, arc=0.08in, boxsep=0pt,
    left=0.15in, right=0.15in, top=0.1in, bottom=0.1in,
    toptitle=0.06in, bottomtitle=0.06in, lefttitle=0.15in, righttitle=0.15in,
  },
  densoftcard/.style={dencard},
  denfillcard/.style={
    denslot, enhanced, colback=#1, colframe=#1, coltext=white,
    boxrule=0pt, arc=0.12in, boxsep=0pt,
    left=0.18in, right=0.18in, top=0.12in, bottom=0.12in,
  },
  denbanner/.style={
    denslot, enhanced, colback=#1, colframe=#1, coltext=white,
    boxrule=0pt, arc=0.06in, boxsep=0pt, valign=center, halign=center,
    left=0.1in, right=0.1in, top=0.02in, bottom=0.02in,
  },
  % Dark banner: dark brown with a light band along the top edge. Drawn with
  % plain fills, not a shading, which PDF viewers and checkers treat as an image.
  denbannershade/.style={
    denslot, enhanced, colback=denbannerbottom, colframe=denbannerbottom, coltext=white,
    boxrule=0pt, arc=0.02in, boxsep=0pt,
    valign=center, halign=center, left=0.1in, right=0.1in, top=0.02in, bottom=0.02in,
    underlay={
      \fill[denbannertop] (frame.north west) rectangle ([yshift=-0.03in]frame.north east);
      \fill[denbannertop!50!denbannerbottom] ([yshift=-0.03in]frame.north west) rectangle ([yshift=-0.07in]frame.north east);
    },
  },
  % An orange disc with an icon, left of the text of a soft card.
  % The text is indented by left=0.95in, set by cols.lua.
  denicondisc/.style={
    overlay={\node[circle, fill=denamber, minimum size=0.62in, inner sep=0pt]
      at ([xshift=0.46in]frame.west) {\includegraphics[width=0.36in, height=0.36in]{#1}};},
  },
  denplain/.style={denslot, blanker},
}
